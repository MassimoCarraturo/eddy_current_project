% Pedagogical plot: Gaussian fusion of a simulation prior with ECT data.
\documentclass[border=2pt]{standalone}
\input{tikz-common.tex}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=86mm, height=56mm,
    xlabel={strain component (arbitrary units)},
    ylabel={probability density},
    domain=-0.7:2.1, samples=200, ymin=0, ymax=2.1,
    legend below columns=2,
    legend style={font=\scriptsize},
    declare function={
      gauss(\x,\m,\s) = exp(-(\x-\m)^2/(2*\s^2))/(\s*sqrt(2*pi));
    },
  ]
  \addplot[gray, thick, densely dashed] {gauss(x,0.8,0.5)};
  \addlegendentry{prior (process simulation)}
  \addplot[oiOrange, thick] {gauss(x,0.2,0.25)};
  \addlegendentry{likelihood (ECT data)}
  \addplot[oiBlue, very thick] {gauss(x,0.32,0.2236)};
  \addlegendentry{posterior (fused estimate)}
\end{axis}
\end{tikzpicture}
\end{document}
