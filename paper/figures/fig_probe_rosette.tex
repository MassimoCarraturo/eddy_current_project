% Pedagogical schematic: directional-probe rosette on the build surface
% (top view), the conductivity analogue of a strain-gauge rosette.
\documentclass[border=2pt]{standalone}
\input{tikz-common.tex}
\begin{document}
\begin{tikzpicture}[font=\small, x=1cm, y=1cm]
  % surface patch
  \fill[gray!8, rounded corners=4pt] (0,0) rectangle (6.4,4.6);
  \draw[gray!60, rounded corners=4pt] (0,0) rectangle (6.4,4.6);
  \node[anchor=south west, gray, font=\footnotesize] at (0.12,0.08)
    {build surface (top view)};
  % probe footprints at 0/60/120 degrees
  \draw[oiBlue,   very thick, fill=oiBlue!12,   fill opacity=0.55]
    (3.2,2.2) ellipse [x radius=1.55, y radius=0.30, rotate=0];
  \draw[oiOrange, very thick, fill=oiOrange!16, fill opacity=0.55]
    (3.2,2.2) ellipse [x radius=1.55, y radius=0.30, rotate=60];
  \draw[oiGreen,  very thick, fill=oiGreen!14,  fill opacity=0.55]
    (3.2,2.2) ellipse [x radius=1.55, y radius=0.30, rotate=120];
  % sensing directions
  \draw[flow, oiBlue,   thick] (3.2,2.2) -- ++(0:2.0)
    node[right, font=\footnotesize] {$\hat{\bm n}_1\,(0^\circ)$};
  \draw[flow, oiOrange, thick] (3.2,2.2) -- ++(60:2.0)
    node[above right, font=\footnotesize] {$\hat{\bm n}_2\,(60^\circ)$};
  \draw[flow, oiGreen,  thick] (3.2,2.2) -- ++(120:2.0)
    node[above left, font=\footnotesize] {$\hat{\bm n}_3\,(120^\circ)$};
  % reading equation
  \node[font=\footnotesize] at (3.2,-0.45)
    {each probe reads
     $\;y_k=\hat{\bm n}_k^{\!\top}\,\bm\sigma\,\hat{\bm n}_k$
     --- three readings fix the in-plane part of $\bm\sigma$};
\end{tikzpicture}
\end{document}
